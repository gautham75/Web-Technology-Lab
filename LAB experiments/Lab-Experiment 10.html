Application Request-Response Lifecycle

How Node.js HTTP server handles incoming form data streams, communicates with MongoDB, and returns dynamically formatted HTML.

1
Form Submission
User submits the registration form with validation to POST /server with URL-encoded payload.
2
Stream Ingestion
Node.js server buffers incoming chunks using req.on('data') and builds rawData until req.on('end').
3
Server Validation
Payload is parsed with URLSearchParams and verified for non-empty required fields and numeric age.
4
MongoDB Insertion
Document is stored in usersCollection.insertOne(newUser) and retrieved via findOne({ _id }).
5
HTML Generation
A formatted HTML response with XSS-escaped table cells is streamed directly back with res.writeHead(200).

Node.js HTTP Server (`Ex10/app.js`)

Pure Node.js implementation utilizing standard http, fs, and official mongodb driver.

Ex10/app.js 
const http = require('http');
const fs = require('fs');
const path = require('path');
const { MongoClient } = require('mongodb');

// Port 7000 is used by default because modern browsers block port 6000 (ERR_UNSAFE_PORT)
const PORT = process.env.PORT || 7000;

// MongoDB connection URL and Database
const mongoUrl = 'mongodb://127.0.0.1:27017';
const dbName = 'mydb';
const collectionName = 'users';

const client = new MongoClient(mongoUrl);

function escapeHTML(str) {
  return String(str ?? '').replace(/[&<>"']/g, char => ({
    '&': '&',
    '<': '<',
    '>': '>',
    '"': '"',
    "'": '''
  })[char]);
}

async function startServer() {
  try {
    await client.connect();
    console.log('Connected successfully to MongoDB server.');
    const db = client.db(dbName);
    const usersCollection = db.collection(collectionName);

    const server = http.createServer(async (req, res) => {
      try {
        if (req.method === 'GET' && req.url === '/') {
          res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
          fs.createReadStream(path.join(__dirname, 'home.html')).pipe(res);
        }
        else if (req.method === 'POST' && req.url === '/server') {
          let rawData = '';
          req.on('data', chunk => { rawData += chunk; });
          req.on('end', async () => {
            const formData = new URLSearchParams(rawData);
            const name = (formData.get('name') || '').trim();
            const password = formData.get('password') || '';
            const age = parseInt(formData.get('age'), 10);
            const mobile = (formData.get('mobile') || '').trim();
            const email = (formData.get('email') || '').trim();
            const gender = formData.get('gender') || '';
            const state = formData.get('state') || '';
            const skills = formData.getAll('skills');

            if (!name || !password || isNaN(age) || !mobile || !email || !gender || !state || skills.length === 0) {
              res.writeHead(400, { 'Content-Type': 'text/html; charset=utf-8' });
              return res.end('<h2 style="color:red; text-align:center;">Validation Error: All fields mandatory!</h2>');
            }

            const newUser = { name, password, age, mobile, email, gender, state, skills, submittedAt: new Date() };
            const insertResult = await usersCollection.insertOne(newUser);
            const savedUser = await usersCollection.findOne({ _id: insertResult.insertedId });

            const rows = [
              ['Name', savedUser.name],
              ['Password', savedUser.password],
              ['Age', savedUser.age],
              ['Mobile Number', savedUser.mobile],
              ['Email', savedUser.email],
              ['Gender', savedUser.gender],
              ['State', savedUser.state],
              ['Skills', Array.isArray(savedUser.skills) ? savedUser.skills.join(', ') : savedUser.skills]
            ];

            const tableHtmlRows = rows.map(([field, value]) => `
              <tr><th>${escapeHTML(field)}</th><td>${escapeHTML(value)}</td></tr>
            `).join('');

            res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
            res.end(`...HTML response with styled table...`);
          });
        }
      } catch (err) {
        res.writeHead(500, { 'Content-Type': 'text/plain' });
        res.end('500 Server Error');
      }
    });

    server.listen(PORT, () => {
      console.log(`Server listening on http://localhost:${PORT}`);
    });
  } catch (err) {
    console.error('Failed to start server:', err);
  }
}

startServer();

Running Experiment 10 Locally

cd Ex10

Step 2: Install MongoDB Driver

npm install

Step 3: Ensure MongoDB Daemon is Running

mongod --dbpath="C:\data\db"

Step 4: Start the Node.js Application Server

node app.js
